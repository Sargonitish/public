-------

#include<stdio.h>
#include<unistd.h>
#include<sys/types.h>
#include<sys/wait.h>

{
    int pid;
    pid = fork();
    if(pid < 0)
    {
        printf("Fork Failed");
    }
    else if(pid == 0)
    {
        printf("Child Process ID : %d\n", getpid());
        printf("Parent Process ID : %d\n", getppid());
        execlp("cal","cal",NULL);
    }
    else
    {
        wait(NULL);
        printf("Parent Process ID : %d\n", getpid());
    }
    return 0;
}

EX.NO : 5.1 FIRST COME FIRST SERVE (FCFS) CPU SCHEDULING
==============================================================================

DATE :

AIM
---

To write a C program to implement the First Come First Serve (FCFS) CPU Scheduling algorithm and generate the Gantt Chart.

ALGORITHM
---------

Step 1: Start the program.
Step 2: Read the number of processes and their burst times.
Step 3: Assign the waiting time of the first process as zero.
Step 4: Calculate the waiting time for the remaining processes.
Step 5: Calculate the turnaround time for each process.
Step 6: Display the Gantt Chart.
Step 7: Calculate the average waiting time and average turnaround time.
Step 8: Stop the program.

PROGRAM
-------

#include<stdio.h>
int main()
{
    int bt[20], wt[20], tat[20];
    int n, i;
    float awt = 0, atat = 0;
    printf("Enter the Number of Processes: ");
    scanf("%d",&n);
    for(i=0;i<n;i++)
    {
        printf("Enter Burst Time for Process P%d: ",i+1);
        scanf("%d",&bt[i]);
    }
    wt[0] = 0;
    for(i=1;i<n;i++)
    wt[i] = wt[i-1] + bt[i-1];
    for(i=0;i<n;i++)
    {
        tat[i] = wt[i] + bt[i];
        awt += wt[i];
        atat += tat[i];
    }
    printf("\nProcess\tBT\tWT\tTAT\n");
    for(i=0;i<n;i++)
    {
        printf("P%d\t%d\t%d\t%d\n",i+1,bt[i],wt[i],tat[i]);
    }
    printf("\nAverage Waiting Time = %.2f",awt/n);
    printf("\nAverage Turnaround Time = %.2f",atat/n);
    return 0;
}

EX.NO : 5.3 ROUND ROBIN (RR) CPU SCHEDULING
==============================================================================

DATE :

AIM
---

To write a C program to implement the Round Robin (RR) CPU Scheduling algorithm and generate the Gantt Chart.

ALGORITHM
---------

Step 1: Start the program.
Step 2: Read the number of processes, burst times, and time quantum.
Step 3: Initialize the remaining burst time of each process.
Step 4: Execute each process for the given time quantum.
Step 5: If the remaining burst time is greater than the time quantum, place the process back in the ready queue.
Step 6: Repeat the process until all processes are completed.
Step 7: Calculate the waiting time and turnaround time for each process.
Step 8: Display the Gantt Chart, average waiting time, and average turnaround time.
Step 9: Stop the program.

PROGRAM
-------

#include<stdio.h>
int main()
{
    int bt[20], rem_bt[20], wt[20], tat[20];
    int n, tq, i, time = 0, remain;
    float awt = 0, atat = 0;
    printf("Enter the Number of Processes: ");
    scanf("%d", &n);
    remain = n;
    for(i = 0; i < n; i++)
    {
        printf("Enter Burst Time for Process P%d: ", i + 1);
        scanf("%d", &bt[i]);
        rem_bt[i] = bt[i];
    }
    printf("Enter Time Quantum: ");
    scanf("%d", &tq);
    while(remain != 0)
    {
        for(i = 0; i < n; i++)
        {
            if(rem_bt[i] > 0)
            {
                if(rem_bt[i] <= tq)
                {
                    time += rem_bt[i];
                    wt[i] = time - bt[i];
                    rem_bt[i] = 0;
                    remain--;
                }
                else
                {
                    rem_bt[i] -= tq;
                    time += tq;
                }
            }
        }
    }
    printf("\nProcess\tBT\tWT\tTAT\n");
    for(i = 0; i < n; i++)
    {
        tat[i] = bt[i] + wt[i];
        awt += wt[i];
        atat += tat[i];
        printf("P%d\t%d\t%d\t%d\n", i + 1, bt[i], wt[i], tat[i]);
    }
    printf("\nAverage Waiting Time = %.2f", awt / n);
    printf("\nAverage Turnaround Time = %.2f", atat / n);
    return 0;
}

EX.NO : 6.1 PRODUCER – CONSUMER PROBLEM
==============================================================================

DATE :

AIM
---

To write a C program to implement the Producer-Consumer problem using semaphores.

ALGORITHM
---------

Step 1: Start the program.
Step 2: Initialize the buffer, mutex, full, and empty semaphores.
Step 3: Read the number of items to be produced.
Step 4: The producer inserts an item into the buffer if it is not full.
Step 5: The consumer removes an item from the buffer if it is not empty.
Step 6: Repeat the producer and consumer operations until all items are processed.
Step 7: Display the producer and consumer operations.
Step 8: Stop the program.

PROGRAM
-------

#include<stdio.h>
int mutex = 1, full = 0, empty = 3, x = 0;
int wait(int s)
{
    return (--s);
}
int signal(int s)
{
    return (++s);
}
int main()
{
    int ch;
    while(1)
    {
        printf("\n1. Producer");
        printf("\n2. Consumer");
        printf("\n3. Exit");
        printf("\nEnter your choice: ");
        scanf("%d",&ch);
        switch(ch)
        {
            case 1:
            if((mutex==1)&&(empty!=0))
            {
                mutex=wait(mutex);
                full=signal(full);
                empty=wait(empty);
                x++;
                printf("\nProducer produces item %d",x);
                mutex=signal(mutex);
            }
            else
            {
                printf("\nBuffer is Full");
            }
            break;
            case 2:
            if((mutex==1)&&(full!=0))
            {
                mutex=wait(mutex);
                full=wait(full);
                empty=signal(empty);
                printf("\nConsumer consumes item %d",x);
                x--;
                mutex=signal(mutex);
            }
            else
            {
                printf("\nBuffer is Empty");
            }
            break;
            case 3:
            return 0;
        }
    }
}

Need = Maximum − Allocation
Step 5: Read the Available resources.
Step 6: Find a process whose Need is less than or equal to the Available resources.
Step 7: Execute the process, release its allocated resources, and update the Available resources.
Step 8: Repeat Steps 6 and 7 until all processes are executed or no safe sequence exists.
Step 9: Display the Safe Sequence if the system is in a safe state; otherwise, display that the system is in an unsafe state.
Step 10: Stop the program.

{
    int alloc[10][10], max[10][10], need[10][10];
    int avail[10], finish[10]={0};
    int safe[10];
    int n, m, i, j, k=0, count=0;
    printf("Enter the number of Processes: ");
    scanf("%d",&n);
    printf("Enter the number of Resources: ");
    scanf("%d",&m);
    printf("Enter Allocation Matrix:\n");
    for(i=0;i<n;i++)
    for(j=0;j<m;j++)
    scanf("%d",&alloc[i][j]);
    printf("Enter Maximum Matrix:\n");
    for(i=0;i<n;i++)
    for(j=0;j<m;j++)
    scanf("%d",&max[i][j]);
    printf("Enter Available Resources:\n");
    for(i=0;i<m;i++)
    scanf("%d",&avail[i]);
    for(i=0;i<n;i++)
    for(j=0;j<m;j++)
    need[i][j]=max[i][j]-alloc[i][j];
    while(count<n)
    {
        int found=0;
        for(i=0;i<n;i++)
        {
            if(finish[i]==0)
            {
                for(j=0;j<m;j++)
                if(need[i][j]>avail[j])
                break;
                if(j==m)
                {
                    for(j=0;j<m;j++)
                    avail[j]+=alloc[i][j];
                    safe[k++]=i;
                    finish[i]=1;
                    found=1;
                    count++;
                }
            }
        }
        if(found==0)
        break;
    }
    if(count==n)
    {
        printf("\nSystem is in Safe State\n");
